warn("loaded")
warn("updated for halloween 2026")
warn("RARELY wont vote")
local players = game:GetService("Players")
local localplayer = players.LocalPlayer

local priority = {
    "Hospital 3",
    "Office 3",
    "House 2",
    "Bank 2",
    "Mansion 2",
    "Workplace",
    "Police Station",
    "Bio Lab",
    "Research Facility",
    "Factory",
    "Mil Base",
    "Hotel 2",
}

local pads = {
    game.Workspace.Lobby.VotePads.VotePad1,
    game.Workspace.Lobby.VotePads.VotePad2,
    game.Workspace.Lobby.VotePads.VotePad3,
}

local hasTPd = false
local lastMaps = {}

local function getText(pad)
    local ok, text = pcall(function()
        return pad.VoteInfoGui.Container.MapName.Text
    end)
    if ok and text ~= "" then
        return text
    end
    return nil
end

for _, pad in next, pads do
    lastMaps[pad] = getText(pad)
end

while true do
    local mapsChanged = false
    for _, pad in next, pads do
        local current = getText(pad)
        if current ~= lastMaps[pad] then
            mapsChanged = true
            lastMaps[pad] = current
        end
    end

    if mapsChanged then
        hasTPd = false
    end

    if not hasTPd then
        for _, mapname in next, priority do
            if hasTPd then break end
            for _, pad in next, pads do
                if lastMaps[pad] == mapname then
                    pcall(function()
                        local character = localplayer.Character
                        if character then
                            local hrp = character:FindFirstChild("HumanoidRootPart")
                            if hrp then
                                hrp.Position = pad.Pad.Position + Vector3.new(0, 5, 0)
                                hrp.Velocity = Vector3.new(0, 0, 0)
                                hasTPd = true
                            end
                        end
                    end)
                end
                if hasTPd then break end
            end
        end
    end

    task.wait(0.25)
end
