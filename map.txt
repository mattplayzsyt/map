local players = game:GetService("Players")
local localplayer = players.LocalPlayer

local priority = {
    "Hospital 3",
    "Office 3",
    "House 2",
    "Bank 2",
    "Mansion 2",
    "Workplace",
    "Police Station",
    "Bio Lab",
    "Research Facility",
    "Factory",
    "Mil Base",
    "Hotel 2",
}

local pads = {
    game.Workspace.RegularLobby.VotePad1,
    game.Workspace.RegularLobby.VotePad2,
    game.Workspace.RegularLobby.VotePad3,
}

local hasTPd = false
local lastMaps = {}

for _, pad in next, pads do
    pcall(function()
        lastMaps[pad] = pad.VoteInfoGui.Container.MapName.Text
    end)
end

while true do
    local mapsChanged = false
    for _, pad in next, pads do
        pcall(function()
            local text = pad.VoteInfoGui.Container.MapName.Text
            if text ~= lastMaps[pad] then
                mapsChanged = true
            end
        end)
    end

    if mapsChanged then
        hasTPd = false
        for _, pad in next, pads do
            pcall(function()
                lastMaps[pad] = pad.VoteInfoGui.Container.MapName.Text
            end)
        end
    end

    if not hasTPd then
        for _, mapname in next, priority do
            if hasTPd then break end
            for _, pad in next, pads do
                pcall(function()
                    local text = pad.VoteInfoGui.Container.MapName.Text
                    if text == mapname then
                        local character = localplayer.Character
                        if character then
                            local hrp = character:FindFirstChild("HumanoidRootPart")
                            if hrp then
                                hrp.Position = pad.Pad.Position + Vector3.new(0, 5, 0)
                                hrp.Velocity = Vector3.new(0, 0, 0)
                                hasTPd = true
                            end
                        end
                    end
                end)
            end
        end
    end

    task.wait(1)
end
